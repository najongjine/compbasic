void * printJumsu(int* jumsus, int len){ // <- 이곳을 구현하시오
    for(int i=0;i<len;i++){
        int s=jumsus[i];
        char* g =(s>=90)?"A":(s>=80)? "B":(s>=70)? "C" : (s>=60) ?"D" :"F";
        printf("%s학점:%03d\n",g,s);
    }

}
